# The editor is compiled INTO mp_audio, not into a library layered above it.
# MasterpieceProcessor::createEditor() is a member of a class declared here, so
# its definition belongs here too: when it lived in mp_ui (which PUBLIC-links
# mp_audio) CMake put mp_ui first on the link line and the linker walked past
# Ui.o before mp_audio introduced the undefined vtable entry. Whole-archiving
# mp_ui "fixed" that by dragging in duplicate copies of JUCE's bundled libpng
# and FLAC. Owning the file outright is the fix; mp_ui stays an INTERFACE
# target so existing link lines keep working.
add_library(mp_audio STATIC
  MasterpieceProcessor.h MasterpieceProcessor.cpp
  SampleLibrary.h SampleLibrary.cpp
  RealtimeThread.h RealtimeThread.cpp

# The score panel (Verovio) is compiled in like the rest of the window; the
# fonts it reads are carried as one zip and unpacked at first use.

  target_sources(mp_audio PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/../mp_ui/ScorePanel.h
    ${CMAKE_CURRENT_SOURCE_DIR}/../mp_ui/ScorePanel.cpp)
  # And the fonts the page's text is drawn with: Bravura for the music
  # symbols, Liberation Serif for the words.
  file(GLOB MP_SCORE_TEXT_FONTS CONFIGURE_DEPENDS
       ${CMAKE_CURRENT_SOURCE_DIR}/../../resources/fonts/*.ttf)
  juce_add_binary_data(mp_score_fonts
    HEADER_NAME MpScoreFonts.h
    NAMESPACE MpScoreFonts
    SOURCES ${MP_SCORE_FONTS_ZIP} ${MP_SCORE_MUSIC_FONT} ${MP_SCORE_TEXT_FONTS})
  target_link_libraries(mp_audio PUBLIC mp_score mp_score_fonts)
  target_compile_definitions(mp_audio PUBLIC MP_HAS_SCORE=1)
endif()

# The program's words in other languages, one plain-text file each. Not
# strings.txt, which is the list translators start from and is never read.
file(GLOB MP_TRANSLATIONS CONFIGURE_DEPENDS
     ${CMAKE_CURRENT_SOURCE_DIR}/../../resources/translations/*.txt)
list(FILTER MP_TRANSLATIONS EXCLUDE REGEX "/strings[.]txt$")
juce_add_binary_data(mp_translations
  HEADER_NAME MpTranslations.h
  NAMESPACE MpTranslations
  SOURCES ${MP_TRANSLATIONS})
